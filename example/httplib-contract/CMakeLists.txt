cmake_minimum_required(VERSION 3.28)
project(httplib_contract_consumer LANGUAGES CXX)
if (NOT VENICE_SOURCE_DIR)
  message(FATAL_ERROR "VENICE_SOURCE_DIR is required")
endif ()
set(${PROJECT_NAME}_INSTALL OFF)

if (PROBE_PREEXISTING)
  find_package(Threads REQUIRED)
  find_package(OpenSSL 3 REQUIRED)
  if (PROBE_PREEXISTING STREQUAL "compiled")
    add_library(httplib::httplib STATIC IMPORTED)
  else ()
    add_library(httplib::httplib INTERFACE IMPORTED)
    set_property(TARGET httplib::httplib PROPERTY INTERFACE_INCLUDE_DIRECTORIES "${PROBE_HEADER_DIR}")
    set_property(TARGET httplib::httplib PROPERTY INTERFACE_COMPILE_DEFINITIONS "${PROBE_DEFINITIONS}")
    if (NOT PROBE_PREEXISTING STREQUAL "missing-tls-link")
      set_property(TARGET httplib::httplib PROPERTY INTERFACE_LINK_LIBRARIES "OpenSSL::SSL;OpenSSL::Crypto;Threads::Threads")
    endif ()
    if (PROBE_PREEXISTING STREQUAL "prefix-aliases")
      add_library(Example::A ALIAS OpenSSL::SSL)
      add_library(Example::AB ALIAS OpenSSL::Crypto)
      set(_alias_links "$<LINK_ONLY:Example::A>;$<LINK_ONLY:Example::AB>;Threads::Threads")
      set_property(TARGET httplib::httplib PROPERTY INTERFACE_LINK_LIBRARIES "${_alias_links}")
    endif ()
  endif ()
  set_property(TARGET httplib::httplib PROPERTY VENICE_SENTINEL retained)
endif ()

include("${VENICE_SOURCE_DIR}/cmake/deps/httplib.cmake")
if (PROBE_PREEXISTING STREQUAL "prefix-aliases")
  get_target_property(_original_links httplib::httplib INTERFACE_LINK_LIBRARIES)
  if (NOT _original_links STREQUAL _alias_links)
    message(FATAL_ERROR "preexisting canonical link interface was changed")
  endif ()
endif ()
if (PROBE_EXPECT_IMPORTED)
  get_target_property(_imported httplib::httplib IMPORTED)
  if (NOT _imported)
    message(FATAL_ERROR "expected installed/imported httplib, but fallback was selected")
  endif ()
endif ()
if (PROBE_PREEXISTING)
  get_target_property(_sentinel httplib::httplib VENICE_SENTINEL)
  if (NOT _sentinel STREQUAL "retained")
    message(FATAL_ERROR "preexisting canonical target was replaced")
  endif ()
endif ()
if (PROBE_SOURCE_SENTINELS)
  foreach (_option IN ITEMS HTTPLIB_TEST HTTPLIB_INSTALL HTTPLIB_COMPILE
      HTTPLIB_BUILD_MODULES HTTPLIB_NO_EXCEPTIONS HTTPLIB_USE_NON_BLOCKING_GETADDRINFO)
    get_property(_cached CACHE ${_option} PROPERTY VALUE)
    if (NOT _cached)
      message(FATAL_ERROR "parent option ${_option} was overwritten")
    endif ()
  endforeach ()
  get_target_property(_type httplib::httplib TYPE)
  if (NOT _type STREQUAL "INTERFACE_LIBRARY")
    message(FATAL_ERROR "embedded dependency is not header-only")
  endif ()
  foreach (_leak IN ITEMS httplib-test httplib_test)
    if (TARGET ${_leak})
      message(FATAL_ERROR "embedded dependency leaked tests")
    endif ()
  endforeach ()
endif ()

add_executable(httplib_contract main.cpp)
target_compile_features(httplib_contract PRIVATE cxx_std_23)
target_include_directories(httplib_contract PRIVATE "${VENICE_SOURCE_DIR}/include")
target_link_libraries(httplib_contract PRIVATE httplib::httplib)
